{{- /* Fetch a Scryfall card by fuzzy name. Use a stable cache key so repeated references to the same card share one cached response. resources.GetRemote handles transient HTTP failures with backoff (including Scryfall's Retry-After on 429s in the Hugo version used by the site), so template-level immediate retries would only add duplicate traffic. Args (dict): name — card name (required) set — set code (optional) Returns: unmarshalled card data, or false on failure. */ -}} {{- $name := .name -}} {{- $set := .set -}} {{- $url := printf "https://api.scryfall.com/cards/named?fuzzy=%s" ($name | urlquery) -}} {{- if $set -}}{{- $url = printf "%s&set=%s" $url ($set | urlquery) -}}{{- end -}} {{- $headers := dict "User-Agent" "hugo-mtg-shortcodes (+https://github.com/jordinebot/hugo-mtg-shortcodes)" "Accept" "application/json" -}} {{- $data := false -}} {{- $opts := dict "headers" $headers "key" $url -}} {{- with try (resources.GetRemote $url $opts) -}} {{- with .Value -}} {{- $data = . | transform.Unmarshal -}} {{- end -}} {{- end -}} {{- return $data -}}
